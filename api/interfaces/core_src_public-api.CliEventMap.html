    Interface CliEventMap

    Payload types of the known events, keyed by event type. The built-in events are listed here; plugins add their own with declaration merging:

    declare module '@qodalis/cli-core' {
        interface CliEventMap {
            'todo:added': { id: number; text: string };
        }
    }

    Events not listed can still be emitted and received; their payload is then unknown.

    interface CliEventMap {
        "boot:completed": {
            commands: number;
            durationMs: number;
            modules: number;
        };
        "command:executed": {
            command: string;
            durationMs: number;
            exitCode: number;
        };
        "config:changed": { category: string; key: string; value: unknown };
        "config:reset": { category: string };
        "module:loaded": { commands: number; name: string; version?: string };
        "module:unloaded": { name: string };
        "package:installed": { name: string; version: string };
        "package:removed": { name: string; version: string };
        "package:updated": {
            name: string;
            previousVersion: string;
            version: string;
        };
        "server:connected": {
            commands: number;
            name: string;
            reconnected: boolean;
            url: string;
        };
        "server:disconnected": { name: string };
        "server:event": { data: unknown; server: string; type: string };
        "server:unavailable": { name: string; url: string };
        "service:completed": { name: string; [key: string]: unknown };
        "service:event": { data: unknown; service: string; type: string };
        "service:failed": { error?: string; name: string; [key: string]: unknown };
        "service:registered": { name: string; [key: string]: unknown };
        "service:started": { name: string; [key: string]: unknown };
        "service:status": { name: string; [key: string]: unknown };
        "service:stopped": { name: string; [key: string]: unknown };
        "session:changed": { user?: string };
    }
