Angular

1.x 
import { CliModule, resolveCommandProcessorProvider } from '@qodalis/angular-cli';
import { CliGuidModule } from '@qodalis/cli-guid';

@NgModule({
    imports: [CliModule, CliGuidModule],
    providers: [resolveCommandProcessorProvider(GreetCommandProcessor)],
})
export class AppModule {}
2.0 
import { CliModule, resolveCliModuleProvider } from '@qodalis/angular-cli';
import { guidModule } from '@qodalis/cli-guid';

@NgModule({
    imports: [CliModule],
    providers: [
        ...resolveCliModuleProvider(guidModule),
        ...resolveCliModuleProvider(myModule),
    ],
})
export class AppModule {}

// or per component: <cli [modules]="[guidModule, myModule]" />
